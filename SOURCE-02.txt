import android.content.Context;
import android.graphics.*;
import android.os.Handler;
import android.view.PixelCopy;

/** Small backdrop samples only while the rail is visible. Works with VideoView's SurfaceView on API 26+. */
final class PlayerGlassRail extends LinearLayout {
    private final Paint paint = new Paint(Paint.ANTI_ALIAS_FLAG | Paint.FILTER_BITMAP_FLAG);
    private final RectF bounds = new RectF(), sampleBounds = new RectF();
    private final Path clip = new Path();
    private final Handler handler = new Handler();
    private SurfaceView source;
    private Bitmap sample;
    private boolean running, copying, disposed;
    private int generation;

    PlayerGlassRail(Context context) {
        super(context);
        setOrientation(HORIZONTAL);
        setGravity(android.view.Gravity.CENTER_VERTICAL);
        setWillNotDraw(false);
        setElevation(0);
    }

    void start(SurfaceView video) {
        source = video;
        if (running || disposed) return;
        running = true;
        handler.post(capture);
    }

    void stop() { running = false; generation++; handler.removeCallbacks(capture); }
    void dispose() { stop(); disposed = true; sample = null; source = null; }

    private final Runnable capture = new Runnable() {
        @Override public void run() {
            if (!running || disposed) return;
            handler.postDelayed(this, 180);
            if (copying || source == null || getWidth() <= 0 || getHeight() <= 0
                    || !source.getHolder().getSurface().isValid()) return;
            int[] rail = new int[2], video = new int[2];
            getLocationOnScreen(rail); source.getLocationOnScreen(video);
            Rect intersect = new Rect(rail[0], rail[1], rail[0] + getWidth(), rail[1] + getHeight());
            if (!intersect.intersect(video[0], video[1], video[0] + source.getWidth(), video[1] + source.getHeight())) {
                if (sample != null) { sample = null; invalidate(); }
                return;
            }
            Rect surface = source.getHolder().getSurfaceFrame();
            float sx = surface.width() / (float) Math.max(1, source.getWidth());
            float sy = surface.height() / (float) Math.max(1, source.getHeight());
            Rect rect = new Rect(Math.max(0, Math.round((intersect.left - video[0]) * sx)),
                    Math.max(0, Math.round((intersect.top - video[1]) * sy)),
                    Math.min(surface.width(), Math.round((intersect.right - video[0]) * sx)),

    private static void blur(Bitmap bitmap) {

        final int radius = 3, size = radius * 2 + 1;
        for (int pass = 0; pass < 2; pass++) {
            for (int y = 0; y < h; y++) for (int x = 0; x < w; x++) {
                int r = 0, g = 0, b = 0;
                for (int k = -radius; k <= radius; k++) {
                    int c = pixels[y * w + Math.max(0, Math.min(w - 1, x + k))];
                    r += (c >> 16) & 255; g += (c >> 8) & 255; b += c & 255;
                }
                temp[y * w + x] = 0xff000000 | ((r / size) << 16) | ((g / size) << 8) | b / size;
            }
            for (int y = 0; y < h; y++) for (int x = 0; x < w; x++) {
                int r = 0, g = 0, b = 0;

                }
                pixels[y * w + x] = 0xff000000 | ((r / size) << 16) | ((g / size) << 8) | b / size;

                    while ((n = in.read(block)) != -1) {
                        if (out.size() + n > 2500000) throw new IOException("response_too_large");
                        out.write(block, 0, n);
                    }

                }

        }

    }
