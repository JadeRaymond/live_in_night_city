package com.reelkeep.app;

import android.content.*;
import android.net.Uri;
import org.json.JSONObject;
import java.io.*;

/** Shares a single existing local video using a temporary read grant, without copying it. */
final class VideoSharing {
    static final String MIME = "video/mp4";
    static Uri uri(Context context, String id) {
        return new Uri.Builder().scheme("content").authority(context.getPackageName()+".videoshare")
                .appendPath(id).appendPath("video.mp4").build();
    }
    static String id(Context context, Uri uri) throws FileNotFoundException {
        if (!"content".equals(uri.getScheme()) || !(context.getPackageName()+".videoshare").equals(uri.getAuthority())
                || uri.getPathSegments().size()!=2 || !"video.mp4".equals(uri.getLastPathSegment())
                || !uri.getPathSegments().get(0).matches("[a-f0-9-]{36}") || uri.getQuery()!=null || uri.getFragment()!=null)
            throw new FileNotFoundException("Unknown video");
        return uri.getPathSegments().get(0);
    }
    static JSONObject item(Context context, Uri uri) throws FileNotFoundException {
        try { return Store.shared(context).find(id(context,uri)); }
        catch(Exception error) { throw new FileNotFoundException("Video unavailable"); }
    }
    static File file(Context context, Uri uri) throws FileNotFoundException {
        File file=MediaDepot.local(context,item(context,uri));
        if(file==null)throw new FileNotFoundException("Download the video first");
        return file;
    }
    static String filename(JSONObject item) {
        String name=item.optString("title","").replaceAll("[^\\p{L}\\p{N} _-]","").trim();
        if(name.length()>70)name=name.substring(0,70);
        return (name.isEmpty()?"Reelkeep-video":name)+".mp4";
    }
    static Intent intent(Context context,String id) throws IOException {
        Uri uri=uri(context,id);file(context,uri);
        Intent intent=new Intent(Intent.ACTION_SEND).setType(MIME).putExtra(Intent.EXTRA_STREAM,uri)
                .addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
        intent.setClipData(ClipData.newRawUri("Reelkeep video",uri));return intent;
    }
}

#!/usr/bin/env bash
set -euo pipefail
cd -- "$(dirname -- "${BASH_SOURCE[0]}")"
BT="$PWD/tools/android-15"
ANDROID="$PWD/tools/android-35/android.jar"
chmod +x "$BT/aapt2" "$BT/zipalign" "$BT/apksigner"
mkdir -p build/classes build/dex
"$BT/aapt2" compile --dir app/res -o build/resources.zip
"$BT/aapt2" link -o build/base.apk --manifest app/AndroidManifest.xml -I "$ANDROID" --java build/generated build/resources.zip -A app/assets
java -jar tools/ecj.jar -8 -proc:none -cp "$ANDROID:tools/ort.jar:tools/tess.jar" -d build/classes app/src/com/reelkeep/app/*.java build/generated/com/reelkeep/app/R.java
java -cp "$BT/lib/d8.jar" com.android.tools.r8.D8 --min-api 26 --lib "$ANDROID" --output build/dex tools/ort.jar tools/tess.jar $(find build/classes -name '*.class')
cp build/base.apk build/unsigned.apk
(cd build/dex && zip -q ../unsigned.apk classes*.dex)
(cd build/native && zip -qr ../unsigned.apk lib)
"$BT/zipalign" -f 4 build/unsigned.apk build/aligned.apk
# Signing material is deliberately external to this review archive.
: "${REELKEEP_KEYSTORE:?Set the path to an external keystore}"
: "${REELKEEP_KS_PASS:?Set the keystore password in the environment}"
